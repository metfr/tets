📘 ШПАРГАЛКА JAVA

⚡ УСЛОВИЯ

if (условие) { ... }

if (age >= 18) {
    System.out.println("Взрослый");
}

if-else

if (score > 70) {
    pass = true;
} else {
    pass = false;
}

else if

if (mark >= 90) grade = 'A';
else if (mark >= 75) grade = 'B';
else grade = 'C';

тернарный оператор (условие ? да : нет)

int max = (a > b) ? a : b;

switch (множественный выбор)

switch (day) {
    case 1: System.out.println("Пн"); break;
    case 2: System.out.println("Вт"); break;
    default: System.out.println("Неизвестно");
}

🔄 ЦИКЛЫ

for (инициализация; условие; шаг)

for (int i = 1; i <= 5; i++) {
    System.out.print(i + " ");
}

while (условие) { ... }

int i = 1;
while (i <= 5) {
    System.out.println(i);
    i++;
}

do-while (выполнится хотя бы раз)

int i = 1;
do {
    System.out.println(i);
    i++;
} while (i <= 5);

вложенные for (таблицы)

for (int i=1; i<=3; i++) {
    for (int j=1; j<=3; j++) {
        System.out.print("*");
    }
    System.out.println();
}
